// ==================== 1. MEDIDAS BASE ====================

// [0 Parámetros] Fecha de corte de la matriz. Formato: dd/mm/yyyy
Fecha Corte =
VAR x0 = CALCULATE ( MAX ( Parametros[Valor] ), Parametros[Parámetro] = "FechaCorte" )
VAR x = DATE ( YEAR ( x0 ), MONTH ( x0 ), DAY ( x0 ) )
RETURN x

// [0 Parámetros] Fecha objetivo de la meta. Formato: dd/mm/yyyy
Fecha Fin Anio =
VAR x0 = CALCULATE ( MAX ( Parametros[Valor] ), Parametros[Parámetro] = "FinAnio" )
VAR x = DATE ( YEAR ( x0 ), MONTH ( x0 ), DAY ( x0 ) )
RETURN x

// [0 Parámetros] Fecha del informe de conciliación (Word). Formato: dd/mm/yyyy
Fecha Informe =
VAR x0 = CALCULATE ( MAX ( Parametros[Valor] ), Parametros[Parámetro] = "FechaInforme" )
VAR x = DATE ( YEAR ( x0 ), MONTH ( x0 ), DAY ( x0 ) )
RETURN x

// [0 Parámetros] Aplicaciones gobernadas objetivo a fin de año. Formato: 0
Meta =
CALCULATE ( MAX ( Parametros[Valor] ), Parametros[Parámetro] = "Meta" )

// [0 Parámetros] Alcance de recepción antes de ampliar criterios. Formato: 0
Alcance Inicial =
CALCULATE ( MAX ( Parametros[Valor] ), Parametros[Parámetro] = "AlcanceInicial" )

// [0 Parámetros] Alcance de recepción con los nuevos criterios. Formato: 0
Alcance Actual =
CALCULATE ( MAX ( Parametros[Valor] ), Parametros[Parámetro] = "AlcanceActual" )

// [0 Parámetros] Gobernadas según el informe de conciliación. Formato: 0
Gobernadas Informe =
CALCULATE ( MAX ( Parametros[Valor] ), Parametros[Parámetro] = "GobernadasInforme" )

// [0 Parámetros] Días promedio por mes. Formato: 0.0
Dias Mes =
CALCULATE ( MAX ( Parametros[Valor] ), Parametros[Parámetro] = "DiasMes" )

// [1 Universo] Aplicaciones registradas en la CMDB. Formato: 0
CMDB =
CALCULATE ( SUM ( dConciliacion[Valor] ), dConciliacion[Tipo] = "Base" )

// [1 Universo] Suma de ajustes de conciliación. Formato: 0
Ajustes CMDB =
CALCULATE ( SUM ( dConciliacion[Valor] ), dConciliacion[Tipo] = "Ajuste" )

// [1 Universo] Universo conciliado de aplicaciones. Formato: 0
Universo =
[CMDB] + [Ajustes CMDB]

// [1 Universo] Filas de la matriz, con duplicados. Formato: 0
Filas Matriz =
COUNTROWS ( fAplicaciones )

// [1 Universo] Aplicaciones sin duplicados exactos (Nombre + Bp Id). Base de todos los conteos. Formato: 0
Aplicaciones Unicas =
CALCULATE ( COUNTROWS ( fAplicaciones ), fAplicaciones[Cuenta] = 1 )

// [1 Universo] Filas repetidas en la matriz. Formato: 0
Duplicados Exactos =
[Filas Matriz] - [Aplicaciones Unicas]

// [1 Universo] Aplicaciones del universo que no están en la matriz. Formato: 0
Sin Fila en Matriz =
MAX ( 0, [Universo] - [Aplicaciones Unicas] )

// [2 Avance] Aplicaciones entregadas a Identidades. Formato: 0
Gobernadas =
CALCULATE ( [Aplicaciones Unicas], fAplicaciones[Gobernada] = 1 )

// [2 Avance] Aplicaciones aún no recibidas. Formato: 0
Pendientes =
CALCULATE ( [Aplicaciones Unicas], fAplicaciones[Gobernada] = 0 )

// [2 Avance] Avance sobre el universo conciliado. Formato: 0%
% Gobernadas Universo =
DIVIDE ( CALCULATE ( [Gobernadas], REMOVEFILTERS ( dEtapa ) ), [Universo] )

// [2 Avance] Cobertura dentro de un segmento (criticidad, equipo, ubicación). Formato: 0%
% Gobernadas Segmento =
DIVIDE ( [Gobernadas], [Aplicaciones Unicas] )

// [2 Avance]  Formato: 0%
% Meta Universo =
DIVIDE ( [Meta], [Universo] )

// [2 Avance]  Formato: 0%
% Meta Cumplida =
DIVIDE ( [Gobernadas], [Meta] )

// [2 Avance] Recepciones que faltan para la meta. Formato: 0
Brecha Meta =
MAX ( 0, [Meta] - [Gobernadas] )

// [2 Avance]  Formato: 0
Criterio 1 =
CALCULATE ( [Aplicaciones Unicas], fAplicaciones[Etapa] = "Criterio 1" )

// [2 Avance]  Formato: 0
Criterio 2 =
CALCULATE ( [Aplicaciones Unicas], fAplicaciones[Etapa] = "Criterio 2" )

// [2 Avance]  Formato: 0
Criterio 3 =
CALCULATE ( [Aplicaciones Unicas], fAplicaciones[Etapa] = "Criterio 3" )

// [2 Avance]  Formato: 0
Planificadas 2027 =
CALCULATE ( [Aplicaciones Unicas], fAplicaciones[Etapa] = "Planificada 2027" )

// [2 Avance] Cartera priorizada para el año. Formato: 0
Priorizadas 2026 =
[Criterio 1] + [Criterio 2] + [Criterio 3]

// [2 Avance] Gobernadas si se recibe todo el criterio 1. Formato: 0
Con Criterio 1 =
[Gobernadas] + [Criterio 1]

// [2 Avance] Del criterio 2 necesarias para la meta. Formato: 0
Necesarias Criterio 2 =
MAX ( 0, [Meta] - [Gobernadas] - [Criterio 1] )

// [3 Ritmo] Recepciones del año de corte. Formato: 0
Recibidas Anio =
VAR c = [Fecha Corte]
RETURN
CALCULATE ( [Gobernadas], REMOVEFILTERS ( dCalendario ),
    fAplicaciones[Fecha Recepcion] >= DATE ( YEAR ( c ), 1, 1 ),
    fAplicaciones[Fecha Recepcion] <= c )

// [3 Ritmo]  Formato: 0.0
Meses Transcurridos =
VAR c = [Fecha Corte]
RETURN DIVIDE ( c - DATE ( YEAR ( c ), 1, 1 ), [Dias Mes] )

// [3 Ritmo] Recepciones por mes en el año. Formato: 0.0
Ritmo Mensual =
DIVIDE ( [Recibidas Anio], [Meses Transcurridos] )

// [3 Ritmo]  Formato: 0.0
Meses Restantes =
DIVIDE ( [Fecha Fin Anio] - [Fecha Corte], [Dias Mes] )

// [3 Ritmo] Recepciones por mes necesarias. Formato: 0.0
Ritmo Requerido =
DIVIDE ( [Brecha Meta], [Meses Restantes] )

// [3 Ritmo]  Formato: 0.0
Multiplicador Ritmo =
DIVIDE ( [Ritmo Requerido], [Ritmo Mensual] )

// [3 Ritmo] Gobernadas a fin de año al ritmo actual. Formato: 0
Proyeccion Fin Anio =
[Gobernadas] + [Ritmo Mensual] * [Meses Restantes]

// [3 Ritmo]  Formato: 0%
% Proyeccion Universo =
DIVIDE ( [Proyeccion Fin Anio], [Universo] )

// [3 Ritmo] Acumulado al final de cada periodo (para un gráfico de líneas con dCalendario[InicioMes]). Incluye las gobernadas sin fecha. Formato: 0
Gobernadas Acumuladas =
VAR c = [Fecha Corte]
VAR f = MAX ( dCalendario[Fecha] )
VAR lim = MIN ( f, c )
RETURN
IF ( MIN ( dCalendario[Fecha] ) > c, BLANK (),
    CALCULATE ( [Gobernadas], REMOVEFILTERS ( dCalendario ),
        FILTER ( ALL ( fAplicaciones[Fecha Recepcion] ),
            fAplicaciones[Fecha Recepcion] <= lim || ISBLANK ( fAplicaciones[Fecha Recepcion] ) ) ) )

// [4 Conciliación]  Formato: 0%
% Informe sobre CMDB =
DIVIDE ( [Gobernadas Informe], [CMDB] )

// [4 Conciliación]  Formato: 0%
% Informe sobre Universo =
DIVIDE ( [Gobernadas Informe], [Universo] )

// [4 Conciliación]  Formato: 0%
% Gobernadas sobre CMDB =
DIVIDE ( CALCULATE ( [Gobernadas], REMOVEFILTERS ( dEtapa ) ), [CMDB] )

// [4 Conciliación] Puntos de más al reportar sobre la CMDB. Formato: 0
Sobrestimacion Puntos =
( [% Gobernadas sobre CMDB] - [% Gobernadas Universo] ) * 100

// [4 Conciliación] Gobernadas con ID propio de Identidades (sin ficha CMDB). Formato: 0
Gobernadas ID GI =
CALCULATE ( [Gobernadas], fAplicaciones[IDPropioGI] = 1 )

// [4 Conciliación]  Formato: 0%
% Gobernadas ID GI =
DIVIDE ( [Gobernadas ID GI], [Gobernadas] )

// [4 Conciliación] Pendientes que ya figuran gobernadas con ID GI. Formato: 0
Posibles Duplicados GI =
CALCULATE ( [Aplicaciones Unicas], fAplicaciones[PosibleDuplicadoGI] = 1 )

// [5 Criticidad] Criticidad A, B o C. Formato: 0
Criticas =
CALCULATE ( [Aplicaciones Unicas], dCriticidad[EsCritica] = 1 )

// [5 Criticidad]  Formato: 0
Criticas Gobernadas =
CALCULATE ( [Gobernadas], dCriticidad[EsCritica] = 1 )

// [5 Criticidad]  Formato: 0%
% Criticas Gobernadas =
DIVIDE ( [Criticas Gobernadas], [Criticas] )

// [5 Criticidad] Críticas diferidas al año siguiente. Formato: 0
Criticas 2027 =
CALCULATE ( [Criticas], fAplicaciones[Etapa] = "Planificada 2027" )

// [5 Criticidad]  Formato: 0
Criticas A 2027 =
CALCULATE ( [Aplicaciones Unicas], dCriticidad[Letra] = "A", fAplicaciones[Etapa] = "Planificada 2027" )

// [5 Criticidad]  Formato: 0
Criterio 1 Baja Criticidad =
CALCULATE ( [Criterio 1], dCriticidad[Letra] IN { "D", "E" } )

// [5 Criticidad]  Formato: 0
Criterio 2 Criticas =
CALCULATE ( [Criterio 2], dCriticidad[EsCritica] = 1 )
